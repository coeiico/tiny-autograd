% \plane{lin|mlp}: the lower-moon region hatched, the 90% lines, the points by class, the
% errors in spot, and the boundary on top.
\newcommand{\plane}[1]{%
  \addplot[draw=none, hatch, empty line=jump] table {build/region-#1.dat};
  \addplot[level line, empty line=jump] table {build/contour-#1-pos.dat};
  \addplot[level line, empty line=jump] table {build/contour-#1-neg.dat};
  \addplot[upper moon] table[x=x, y=y, restrict expr to domain={\thisrow{cls}}{0:0}] {build/points-#1.dat};
  \addplot[lower moon] table[x=x, y=y, restrict expr to domain={\thisrow{cls}}{1:1}] {build/points-#1.dat};
  \addplot[marked] table[x=x, y=y, restrict expr to domain={\thisrow{wrong}}{1:1}] {build/points-#1.dat};
  \addplot[draw=ink, line width=0.9pt, empty line=jump] table {build/contour-#1-zero.dat};
}
